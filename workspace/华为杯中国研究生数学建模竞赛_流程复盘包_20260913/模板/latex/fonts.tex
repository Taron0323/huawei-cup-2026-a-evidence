% Fonts remain in their installed location; no proprietary font is redistributed.
\newcommand{\modelcjkfrompath}[1]{%
  \IfFileExists{#1Kaiti.ttf}{%
    \setCJKmainfont{Simsun.ttc}[Path={#1},FontIndex=0,AutoFakeBold=2.2,ItalicFont=Kaiti.ttf]
  }{%
    \setCJKmainfont{Simsun.ttc}[Path={#1},FontIndex=0,AutoFakeBold=2.2,AutoFakeSlant=0.2]
  }%
  \setCJKfamilyfont{zhsong}{Simsun.ttc}[Path={#1},FontIndex=0,AutoFakeBold=2.2]
  \typeout{MODELFONTS-SONG: SimSun}
  \IfFileExists{#1SimHei.ttf}{%
  \setCJKsansfont{SimHei.ttf}[Path={#1},AutoFakeBold=1.5]
  \setCJKfamilyfont{zhhei}{SimHei.ttf}[Path={#1},AutoFakeBold=1.5]
  \typeout{MODELFONTS-HEI: SimHei}%
  }{\typeout{MODELFONTS-HEI: FandolHei fallback}}%
  \IfFileExists{#1Kaiti.ttf}{%
  \setCJKfamilyfont{zhkai}{Kaiti.ttf}[Path={#1},AutoFakeBold=2]
  \typeout{MODELFONTS-KAI: KaiTi}%
  }{\typeout{MODELFONTS-KAI: FandolKai fallback}}}

\ifmodelportablefonts
  \typeout{MODELFONTS-CJK: Fandol portable mode}
\else
\IfFileExists{fonts/Simsun.ttc}{%
  \modelcjkfrompath{fonts/}%
}{%
  \IfFileExists{/Applications/Microsoft Word.app/Contents/Resources/DFonts/Simsun.ttc}{%
    \modelcjkfrompath{/Applications/Microsoft Word.app/Contents/Resources/DFonts/}%
  }{%
    \IfFontExistsTF{SimSun}{%
      \setCJKmainfont{SimSun}[AutoFakeBold=2.2]
      \setCJKfamilyfont{zhsong}{SimSun}[AutoFakeBold=2.2]
      \IfFontExistsTF{SimHei}{\setCJKsansfont{SimHei}\setCJKfamilyfont{zhhei}{SimHei}}{}
      \IfFontExistsTF{KaiTi}{\setCJKfamilyfont{zhkai}{KaiTi}[AutoFakeBold=2]}{}
      \typeout{MODELFONTS-CJK: system fonts, verify individual family availability}%
    }{\typeout{MODELFONTS-CJK: Fandol fallback, not identical to the reference}}%
  }%
}
\fi
